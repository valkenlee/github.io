/* 대기 유형(단기, 샤보, 간짱, 변짱, 양면) 정밀 판별 함수 */ function classifyWaitType(decomp, addedTile, originalHand) { // 1. 단기대기: 완성 패가 머리(작두)가 되는 경우 if (decomp.pair === addedTile) { return { waitType: '단기', targetMeldStart: null }; } let origCounts = Array(10).fill(0); originalHand.forEach(n => origCounts[n]++); // 2. 샤보대기: 원본 손패에 이미 2장(쌍)이 있었고 완성되어 커츠가 되는 경우 if (decomp.triplets.includes(addedTile) && origCounts[addedTile] === 2) { return { waitType: '샤보', targetMeldStart: null }; } // 3. 슌츠 완성 대기 (변짱, 간짱, 양면) for (let s of decomp.sequences) { if (addedTile >= s && addedTile <= s + 2) { const pos = addedTile - s; // 간짱 대기: 슌츠의 가운데 패 (예: 1,3에서 2 대기) if (pos === 1) { return { waitType: '간짱', targetMeldStart: s }; } // 변짱 대기: 정확히 1,2에서 3 대기(s=1, added=3) 또는 8,9에서 7 대기(s=7, added=7) if ((s === 1 && addedTile === 3) || (s === 7 && addedTile === 7)) { return { waitType: '변짱', targetMeldStart: s }; } // 양면 대기: 변짱 조건에 해당하지 않는 모든 양 끝 대기 (예: 2,3,4에서 1 대기) if (pos === 0 || pos === 2) { return { waitType: '양면', targetMeldStart: s }; } } } return { waitType: '양면', targetMeldStart: null }; }
